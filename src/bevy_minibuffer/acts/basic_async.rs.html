24/// Execute an act by name. Similar to Emacs' `M-x` or vim's `:` key binding.
25pub fn run_act(
26    mut minibuffer: MinibufferAsync,
27    mut act_cache: ResMut<NameActCache>,
28    mut acts: Query<(Entity, &Act)>,
29    last_act: Res<LastRunAct>,
30) -> impl Future<Output = Result<(), crate::Error>> {
31    let acts_trie = act_cache
32        .trie(acts.iter(), ActFlags::RunAct | ActFlags::Active)
33        .clone();
34    let prompt: Cow<'static, str> = last_act
35        .hotkey(&mut acts.transmute_lens::<&Act>())
36        .map(|hotkey| {
37            // We're hardcoding this little vim-ism. We feel slightly vandalous
38            // _and_ good about it.
39            if hotkey.alias.as_ref().map(|x| x == ":").unwrap_or(false) {
40                // All it does is remove the space after the prompt.
41                ":".into()
42            } else {
43                format!("{} ", hotkey).into()
44            }
45        })
46        .unwrap_or("run_act: ".into());
47    async move {
48        match minibuffer.prompt_map(prompt, acts_trie).await {
49            // TODO: Get rid of clone.
50            Ok(act_ref) => {
51                AsyncWorld::new().write_message(RunActEvent::new(act_ref))?;
52            }
53            Err(e) => {
54                minibuffer.message(format!("Error: {e}"));
55            }
56        }
57        Ok(())
58    }
59}

61/// Input a key sequence. This will tell you what it does.
62pub fn describe_key(
63    acts: Query<(Entity, &Act)>,
64    mut cache: ResMut<HotkeyActCache>,
65    mut minibuffer: MinibufferAsync,
66) -> impl Future<Output = Result<(), crate::Error>> {
67    use trie_rs::inc_search::Answer;
68    let act_names: HashMap<Entity, Cow<'static, str>> = acts
69        .iter()
70        .map(|(id, act)| (id, act.name.clone()))
71        .collect();
72    let trie: Trie<_, _> = cache.trie(acts.iter()).clone();
73    async move {
74        let mut search = trie.inc_search();
75        let prompt = "Press key: ";
76        let mut accum = String::new();
77        loop {
78            minibuffer.message(format!("{}{}", prompt, accum));
79            let chord = minibuffer.get_chord().await?;
80            match search.query(&chord) {
81                Some(x) => {
82                    let _ = write!(accum, "{} ", chord);
83                    let name: Option<&Cow<'static, str>> = search
84                        .value()
85                        .and_then(|act_ref: &ActRef| act_names.get(&act_ref.id));
86                    let msg = match x {
87                        Answer::Match => {
88                            format!("{}is bound to {}", accum, name.unwrap_or(&"???".into()))
89                        }
90                        Answer::PrefixAndMatch => {
91                            format!(
92                                "{}is bound to {} and more",
93                                accum,
94                                name.unwrap_or(&"???".into())
95                            )
96                        }
97                        Answer::Prefix => accum.clone(),
98                    };
99                    minibuffer.message(msg);
100                    if matches!(x, Answer::Match) {
101                        break;
102                    }
103                }
104                None => {
105                    let _ = write!(accum, "{} ", chord);
106                    let msg = format!("{}is unbound", accum);
107                    minibuffer.message(msg);
108                    break;
109                }
110            }
111        }
112        Ok(())
113    }
114}
115/// Bare minimum of acts for a useable and discoverable console
116///
117/// Key bindings may be altered or removed prior to adding this as a
118/// plugin. Likewise acts may be altered or removed.
119///
120/// Although it is a [Plugin], if you use [App::add_plugins], the acts will not
121/// be added. [ActBuilder] contains a non-cloneable that must be taken which
122/// [Plugin::build] does not permit with its read-only `&self` access. Instead
123/// use [AddActs::add_acts].
124#[derive(Debug, Deref, DerefMut)]
125pub struct BasicAsyncActs {
126    /// Set of basic acts
127    pub acts: Acts,
128}
129
130impl BasicAsyncActs {
131    /// Substitute acts within [BasicActs].
132    pub fn subtitute_async(mut self, basic_acts: &mut BasicActs) {
133        let mut last_name = String::new();
134        for (name, act) in self.take_acts().0.into_iter() {
135            last_name.replace_range(.., &name);
136            if basic_acts.insert(name, act).is_none() {
137                warn!("Substitution of act '{}' not present.", last_name);
138            }
139        }
140    }
141}
142
143impl Default for BasicAsyncActs {
144    fn default() -> Self {
145        Self {
146            acts: Acts::new([
147                ActBuilder::new(run_act.pipe(sink::future_result))
148                    .named("run_act")
149                    .bind_aliased(keyseq! { Shift-; }, ":")
150                    .bind(keyseq! { Alt-X })
151                    .add_flags(ActFlags::Adverb)
152                    .sub_flags(ActFlags::RunAct),
153                ActBuilder::new(describe_key.pipe(sink::future_result))
154                    .named("describe_key")
155                    .bind(keyseq! { Ctrl-H K }),
156            ]),
157        }
158    }
159}

161impl Plugin for BasicAsyncActs {
162    fn build(&self, _app: &mut App) {
163        self.warn_on_unused_acts();
164    }
165}

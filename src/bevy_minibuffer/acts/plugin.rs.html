6/// A [Plugin] with a collection of [Acts]
7///
8/// Although it is a [Plugin], if you use [App::add_plugins] with it, then its
9/// acts will not be added. [ActBuilder] contains a non-cloneable field that
10/// must be taken. [Plugin::build] does not permit this with its read-only
11/// `&self` access. Instead we use [AddActs::add_acts] to add both the acts and
12/// the Plugin comprising an ActsPlugin.
13pub trait ActsPlugin: Plugin {
14    /// Immutable reference to acts.
15    fn acts(&self) -> &Acts;
16    /// Mutable reference to acts.
17    fn acts_mut(&mut self) -> &mut Acts;
18
19    /// Take the acts. This removes them from the plugin so that they may be
20    /// altered and the plugin may then be added with [App::add_plugins] or
21    /// [AddActs::add_acts].
22    fn take_acts(&mut self) -> Acts {
23        self.acts_mut().take()
24    }
25
26    /// Warn if there are unused acts in this `ActsPlugin`.
27    ///
28    /// Typically this should be called in the implementers `Plugin::build()`
29    /// method. This will provide some safeguard to ensure `Act`s don't get lost
30    /// in the shuffle.
31    fn warn_on_unused_acts(&self) {
32        let acts = self.acts();
33        if !acts.is_empty() {
34            let count = acts.len();
35            warn!(
36                "{} has {} act{} not added; consider using add_acts() instead of add_plugins() for it.",
37                self.name(),
38                count,
39                if count == 1 { " that was" } else { "s that were" }
40            );
41        }
42    }
43
44    /// Configure an act.
45    ///
46    /// Will panic if no act with that name is available.
47    fn configure<F: Fn(&mut ActBuilder)>(mut self, act_name: &str, f: F) -> Self
48    where
49        Self: Sized,
50    {
51        if let Some(ref mut builder) = self.acts_mut().get_mut(act_name) {
52            f(builder);
53        } else {
54            panic!("No act named {:?} to configure.", act_name);
55        }
56        self
57    }
58}

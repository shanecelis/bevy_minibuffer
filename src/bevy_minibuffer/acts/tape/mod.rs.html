297    fn play_for(
298        mut query: Query<(Entity, &AudioSink, &mut PlayFor), With<TapeSoundSink>>,
299        mut commands: Commands,
300        time: Res<Time>,
301        mut tape_state: ResMut<NextState<SoundState>>,
302    ) {
303        for (id, _sink, mut play_for) in &mut query {
304            let PlayFor(ref mut timer, ref mut after) = *play_for;
305            timer.tick(time.delta());
306            if timer.just_finished() {
307                match std::mem::take(after) {
308                    After::Done => {
309                        warn!("After::DONE unexpected.");
310                    }
311                    After::Play(bundle) => {
312                        commands.spawn((bundle.source, bundle.settings, TapeSoundSink));
313                        commands.entity(id).despawn();
314                    }
315                    After::State(state) => {
316                        tape_state.set(state);
317                        commands.entity(id).despawn();
318                    }
319                }
320            }
321        }
322    }

324    fn after(
325        mut query: Query<(Entity, &AudioSink, &mut After), With<TapeSoundSink>>,
326        mut commands: Commands,
327        mut tape_state: ResMut<NextState<SoundState>>,
328    ) {
329        for (id, sink, mut after) in &mut query {
330            if sink.empty() {
331                match std::mem::take(&mut *after) {
332                    After::Done => {
333                        warn!("After::DONE unexpected.");
334                    }
335                    After::Play(bundle) => {
336                        commands.spawn((bundle.source, bundle.settings, TapeSoundSink));
337                        commands.entity(id).despawn();
338                    }
339                    After::State(state) => {
340                        tape_state.set(state);
341                        commands.entity(id).despawn();
342                    }
343                }
344            }
345        }
346    }

348    fn animate_icon(
349        mut query: Query<&mut Transform, With<TapeIcon>>,
350        mut animate: ResMut<TapeAnimate>,
351        last_speed: Local<Option<f32>>,
352        time: Res<Time>,
353    ) {
354        let speed = match *animate {
355            TapeAnimate::Speed(speed) => Some(speed),
356            TapeAnimate::Hide => None,
357            TapeAnimate::Curve {
358                ref curve,
359                ref mut pos,
360            } => {
361                let r = curve.sample(*pos);
362                *pos += time.delta_secs();
363                // This could be none. Should do something at that point.
364                r.or_else(|| *last_speed)
365            }
366        };
367        // *last_speed = speed;
368
369        if let Some(speed) = speed {
370            for mut transform in &mut query {
371                transform.rotate_local_z(speed * time.delta_secs());
372            }
373        }
374    }

511pub(crate) fn process_event(
512    mut events: MessageReader<RunActEvent>,
513    mut recorder: ResMut<TapeRecorder>,
514    universal_arg: Res<UniversalArg>,
515) {
516    for event in events.read() {
517        if event.act.flags.contains(ActFlags::Record) {
518            match *recorder {
519                TapeRecorder::Off {
520                    one_off: ref mut tape,
521                } => {
522                    tape.content.clear();
523                    tape.append_run(event, &universal_arg);
524                }
525                TapeRecorder::Record { ref mut tape, .. } => {
526                    tape.append_run(event, &universal_arg);
527                }
528                _ => (),
529            }
530        }
531    }
532}

565fn tape_record(
566    mut minibuffer: Minibuffer,
567    mut tapes: ResMut<Tapes>,
568    universal: Res<UniversalArg>,
569    mut next_tape_state: ResMut<NextState<SoundState>>,
570) {
571    next_tape_state.set(SoundState::Load);
572    let append = universal.is_some();
573    match &*minibuffer.tape_recorder {
574        TapeRecorder::Off { one_off: _ } => {
575            minibuffer.message("Record tape: ");
576            minibuffer.get_chord().observe(
577                move |mut trigger: On<KeyChordEvent>,
578                      tapes: Res<Tapes>,
579                      mut commands: Commands,
580                      mut minibuffer: Minibuffer,
581                      mut tape_state: ResMut<NextState<SoundState>>| {
582                    let entity = trigger.event().entity;
583                    match trigger.event_mut().take() {
584                        Ok(chord) => {
585                            tape_state.set(SoundState::Record);
586                            if append {
587                                if let Some(tape) = tapes.get(&chord) {
588                                    minibuffer.message(format!("Recording tape {}", chord));
589                                    *minibuffer.tape_recorder = TapeRecorder::Record {
590                                        tape: tape.clone(),
591                                        chord,
592                                    };
593                                } else {
594                                    minibuffer.message(format!(
595                                        "No prior tape. Recording new tape {}",
596                                        chord
597                                    ));
598                                    *minibuffer.tape_recorder = TapeRecorder::Record {
599                                        tape: Tape::default(),
600                                        chord,
601                                    };
602                                }
603                            } else {
604                                minibuffer.message(format!("Recording new tape {}", chord));
605                                *minibuffer.tape_recorder = TapeRecorder::Record {
606                                    tape: Tape::default(),
607                                    chord,
608                                };
609                            }
610                        }
611                        Err(e) => {
612                            tape_state.set(SoundState::Squeak);
613                            minibuffer.message(format!("{e}"));
614                        }
615                    }
616                    commands.entity(entity).despawn();
617                },
618            );
619        }
620        TapeRecorder::Record { .. } => {
621            let TapeRecorder::Record { tape, chord } =
622                std::mem::take(&mut *minibuffer.tape_recorder)
623            else {
624                unreachable!();
625            };
626            next_tape_state.set(SoundState::Stop);
627            minibuffer.message(format!("Stop recording tape {}", chord));
628            tapes.insert(chord, tape);
629        }
630        TapeRecorder::Play => {
631            warn!("Got record tape during Play.");
632        }
633    }
634}

636#[allow(clippy::too_many_arguments)]
637fn tape_play(
638    In(chord): In<Option<KeyChord>>,
639    mut minibuffer: Minibuffer,
640    mut acts: Query<&Act>,
641    last_act: Res<LastRunAct>,
642    universal_arg: Res<UniversalArg>,
643    tapes: Res<Tapes>,
644    mut next_tape_state: ResMut<NextState<SoundState>>,
645    tape_state: Res<State<SoundState>>,
646    mut commands: Commands,
647) {
648    // Non-interactive case
649    if let Some(chord) = chord {
650        if let Some(tape) = tapes.get(&chord) {
651            let tape = tape.clone();
652            let count = 1; // TODO: Need to store universal somewhere.
653                           // next_tape_state.set(SoundState::Play);
654            commands.queue(move |world: &mut World| {
655                for _ in 0..count {
656                    if let Err(e) = world.run_system_cached_with(play_tape_sys, &tape) {
657                        warn!("Error playing tape: {e:?}");
658                    }
659                }
660            });
661        } else {
662            // next_tape_state.set(SoundState::Load);
663            minibuffer.message(format!("No tape for {}", chord));
664        }
665        return;
666    }
667    // Interactive case
668    if *tape_state.get() != SoundState::Play {
669        next_tape_state.set(SoundState::Load);
670    }
671    let this_keychord = last_act.hotkey(&mut acts.as_query_lens());
672    let count = universal_arg.unwrap_or(1);
673    minibuffer.message("Play tape: ");
674    minibuffer.get_chord().observe(
675        move |mut trigger: On<KeyChordEvent>,
676              mut commands: Commands,
677              tapes: Res<Tapes>,
678              mut minibuffer: Minibuffer,
679              mut tape_state: ResMut<NextState<SoundState>>,
680              mut last_played: ResMut<LastPlayed>| {
681            let entity = trigger.event().entity;
682            match trigger.event_mut().take() {
683                Ok(mut chord) => 'body: {
684                    if this_keychord
685                        .as_ref()
686                        .map(|x| x.chords.len() == 1 && x.chords[0] == chord)
687                        .unwrap_or(false)
688                    {
689                        // We want to play the same chord as last time.
690                        if let Some(ref last_played) = **last_played {
691                            chord.clone_from(last_played);
692                        } else {
693                            minibuffer
694                                .message("Tried to play last tape but no tape has been played.");
695                            break 'body;
696                        }
697                    }
698                    minibuffer.log_input(&Some(chord));
699                    if let Some(tape) = tapes.get(&chord) {
700                        let tape = tape.clone();
701                        tape_state.set(SoundState::Play);
702                        commands.queue(move |world: &mut World| {
703                            for _ in 0..count {
704                                if let Err(e) = world.run_system_cached_with(play_tape_sys, &tape) {
705                                    warn!("Error playing tape: {e:?}");
706                                }
707                            }
708                        });
709                        **last_played = Some(chord);
710                    } else {
711                        tape_state.set(SoundState::Load);
712                        minibuffer.message(format!("No tape for {}", chord));
713                    }
714                }
715                Err(e) => {
716                    minibuffer.message(format!("{e}"));
717                    tape_state.set(SoundState::Squeak);
718                }
719            }
720            commands.entity(entity).despawn();
721        },
722    );
723}

725fn tape_copy(mut minibuffer: Minibuffer, mut tape_state: ResMut<NextState<SoundState>>) {
726    tape_state.set(SoundState::Load);
727    minibuffer.message("Copy tape for key: ");
728    minibuffer.get_chord().observe(
729        |mut trigger: On<KeyChordEvent>,
730         mut commands: Commands,
731         tapes: Res<Tapes>,
732         mut minibuffer: Minibuffer,
733         run_act_map: Res<RunActMap>,
734         mut tape_state: ResMut<NextState<SoundState>>,
735         acts: Query<&Act>| {
736            let entity = trigger.event().entity;
737            match trigger.event_mut().take() {
738                Ok(chord) => 'press: {
739                    if let Some(tape) = tapes.get(&chord) {
740                        let script = match tape.script(&acts, &run_act_map) {
741                            Ok(s) => s,
742                            Err(e) => {
743                                minibuffer.message(format!("Could not generate script: {e}"));
744                                break 'press;
745                            }
746                        };
747                        tape_state.set(SoundState::Rewind);
748                        info!("{}", script);
749                        #[cfg(feature = "clipboard")]
750                        {
751                            match ClipboardContext::new() {
752                                Ok(mut ctx) => {
753                                    if let Err(e) = ctx.set_contents(script.clone()) {
754                                        warn!("Could not set clipboard: {e}");
755                                    }
756                                    minibuffer.message(format!(
757                                        "Copy tape {} to clipboard and log:\n\n{}.",
758                                        &chord, &script
759                                    ));
760                                }
761                                Err(e) => {
762                                    minibuffer
763                                        .message(format!("Log tape {}:\n\n{}", &chord, &script));
764                                    warn!("Could not initialize clipboard: {e}");
765                                }
766                            }
767                        }
768                        #[cfg(not(feature = "clipboard"))]
769                        minibuffer.message(format!("Log tape {}:\n\n{}", chord, script));
770                    } else {
771                        minibuffer.message(format!("No tape for {}", chord));
772                    }
773                }
774                Err(e) => {
775                    tape_state.set(SoundState::Squeak);
776                    minibuffer.message(format!("{e}"));
777                }
778            }
779            commands.entity(entity).despawn();
780        },
781    );
782}

784fn repeat(
785    tape_recorder: Res<TapeRecorder>,
786    universal_arg: Res<UniversalArg>,
787    mut commands: Commands,
788) {
789    let count = universal_arg.unwrap_or(1);
790    if let TapeRecorder::Off { ref one_off } = *tape_recorder {
791        let tape = one_off.clone();
792        commands.queue(move |world: &mut World| {
793            for _ in 0..count {
794                if let Err(e) = world.run_system_cached_with(play_tape_sys, &tape) {
795                    warn!("Error playing tape: {e:?}");
796                }
797            }
798        });
799    }
800}

802impl Tape {
803    pub fn append_run(&mut self, act: &RunActEvent, universal_arg: &UniversalArg) {
804        self.content
805            .push(RunActRecord::from(act).with_universal(universal_arg));
806    }
807
808    pub fn ammend_input<I: Clone + 'static + Send + Sync + Debug>(&mut self, input: I) {
809        if let Some(ref mut entry) = self.content.last_mut() {
810            if entry.input.is_some() {
811                warn!("Overwriting tape input for act {}", &entry.act.id);
812            }
813            entry.input = Some(Arc::new(input));
814        } else {
815            warn!("Cannot append input; no act has been run.");
816        }
817    }
818
819    fn script(&self, acts: &Query<&Act>, run_act_map: &RunActMap) -> Result<String, fmt::Error> {
820        let mut f = String::new();
821        writeln!(f, "fn tape(mut commands: Commands) {{")?;
822        for event in &self.content {
823            let Ok(act) = acts.get(event.act.id) else {
824                continue;
825            };
826            match event.input {
827                Some(ref input) => {
828                    let type_id = (**input).type_id();
829                    // info!("try to get type_id out {type_id:?}");
830                    let input_string: Cow<'static, str> = match run_act_map.get(&type_id) {
831                        Some(run_act) => match run_act.debug_string(&**input) {
832                            Some(s) => s.into(),
833                            None => {
834                                warn!("Debug string function failed for act {:?}", &act.name);
835                                "???".into()
836                            }
837                        },
838                        None => {
839                            warn!("No debug string function for act {:?}", &act.name);
840                            "!!!".into()
841                        }
842                    };
843                    writeln!(
844                        f,
845                        "    commands.run_system_cached_with({}, {})",
846                        act.system_name, input_string
847                    )?;
848                }
849                None => {
850                    writeln!(f, "    commands.run_system_cached({})", act.system_name)?;
851                }
852            }
853        }
854        write!(f, "}}")?;
855        Ok(f)
856    }
857}

876pub fn play_tape_sys(
877    InRef(tape): InRef<Tape>,
878    mut commands: Commands,
879    mut tape_recorder: ResMut<TapeRecorder>,
880    acts: Query<&Act>,
881    run_act_map: Res<crate::acts::RunActMap>,
882    mut universal_arg: ResMut<UniversalArg>,
883) {
884    let count = universal_arg.take().unwrap_or(1);
885    let old = std::mem::replace(&mut *tape_recorder, TapeRecorder::Play);
886    for _ in 0..count {
887        for e in &tape.content {
888            let Ok(act) = acts.get(e.act.id) else {
889                warn!("Could not get act for {:?}", e.act.id);
890                // XXX: Shouldn't I break here?
891                continue;
892            };
893            let run_act = act
894                .input
895                .as_ref()
896                .and_then(|x| run_act_map.get(x).map(|y| &**y));
897
898            let run_act = run_act.unwrap_or(&ActSystem);
899            *universal_arg = e.universal.clone();
900            if let Some(ref input) = e.input {
901                let input = input.clone();
902                if let Err(error) = run_act.run_with_input(act.system_id, &*input, &mut commands) {
903                    warn!("Error running act with input '{}': {:?}", act.name, error);
904                }
905            } else if let Err(error) = run_act.run(act.system_id, &mut commands) {
906                warn!("Error running act '{}': {:?}", act.name, error);
907            }
908        }
909    }
910    let _ = std::mem::replace(&mut *tape_recorder, old);
911}

21/// Prompt to get one-line user input.
22///
23/// # Key Events
24///
25/// | Key         | Action                       |
26/// | ----------- | ---------------------------- |
27/// | `Enter`     | Submit current/initial value |
28/// | `Backspace` | Delete previous character    |
29/// | `Delete`    | Delete current character     |
30/// | `Left`      | Move cursor left             |
31/// | `Right`     | Move cursor right            |
32///
33#[derive(Component, Deref)]
34// #[reflect(opaque)]
35pub(crate) struct AutoComplete(Box<dyn Lookup + Send + Sync>);
36
37/// Means that an auto completing read must match one of its lookups.
38#[derive(Component, Debug)]
39pub struct RequireMatch;
40
41impl AutoComplete {
42    /// Wrap a prompt in autocomplete.
43    pub fn new<L>(lookup: L) -> Self
44    where
45        L: Lookup + Send + Sync + 'static,
46    {
47        Self(Box::new(lookup))
48    }
49
50    /// Construct an autocomplete UI element.
51    ///
52    /// NOTE: Could not use the `Construct` trait.
53    pub fn construct(
54        self,
55        mut commands: EntityCommands,
56        prompt: impl Into<Cow<'static, str>>,
57    ) -> EntityCommands {
58        commands
59            .insert(Prompt(prompt.into()))
60            .insert(Node {
61                flex_wrap: FlexWrap::Wrap,
62                ..default()
63            })
64            .insert(StringCursor::default())
65            .insert(next_tab_index())
66            .construct::<crate::view::View>(())
67            .insert(self);
68        commands
69    }
70}

106fn autocomplete_controller(
107    mut input_focus: ResMut<InputFocus>,
108    nav: TabNavigation,
109    mut query: Query<(
110        Entity,
111        &mut StringCursor,
112        &AutoComplete,
113        Option<&RequireMatch>,
114    )>,
115    mut input: MessageReader<KeyboardInput>,
116    mut commands: Commands,
117    mut lookup_events: MessageWriter<LookupEvent>,
118    frame_count: Res<FrameCount>,
119) {
120    // Always read so the reader drains each frame; otherwise messages persist and
121    // the trigger key (e.g. ':') can appear in the next frame when the prompt opens.
122    for ev in input.read() {
123        if ev.state != ButtonState::Pressed {
124            continue;
125        }
126        for (id, mut text_state, autocomplete, require_match) in query.iter_mut() {
127            if input_focus.get() != Some(id) {
128                continue;
129            }
130            trace!("input {:?} frame {}", &ev.logical_key, frame_count.0);
131            match &ev.logical_key {
132                Key::Tab => {
133                    if let Err(e) = autocomplete.lookup(&text_state.value) {
134                        use LookupError::*;
135                        match e {
136                            NoMatch => {
137                                lookup_events.write(LookupEvent::Hide);
138                                if let Ok(mut ecommands) = commands.get_entity(id) {
139                                    ecommands.try_insert(Feedback::info(format!("{}", e)));
140                                }
141                            }
142                            OneMatch(s) => {
143                                lookup_events.write(LookupEvent::Hide);
144                                text_state.set_value(&s);
145                            }
146                            ManyMatches => {
147                                let matches = autocomplete.all_lookups(&text_state.value);
148
149                                lookup_events.write(LookupEvent::Completions(matches));
150                                if let Some(new_input) =
151                                    autocomplete.longest_prefix(&text_state.value)
152                                {
153                                    text_state.set_value(&new_input);
154                                }
155                            }
156                            Message(s) => {
157                                lookup_events.write(LookupEvent::Hide);
158                                if let Ok(mut ecommands) = commands.get_entity(id) {
159                                    ecommands.try_insert(Feedback::info(s)); // Err(s),
160                                }
161                            }
162                        }
163                    }
164                }
165                Key::Character(s) => {
166                    for c in s.chars() {
167                        text_state.insert(c);
168                    }
169                }
170                Key::Space => text_state.insert(' '),
171                Key::Backspace => text_state.backspace(),
172                Key::Delete => text_state.delete(),
173                Key::ArrowLeft => text_state.move_cursor(CursorDirection::Left),
174                Key::ArrowRight => text_state.move_cursor(CursorDirection::Right),
175                Key::Enter => {
176                    if require_match.is_some() {
177                        if let Err(e) = autocomplete.lookup(&text_state.value) {
178                            use LookupError::*;
179                            match e {
180                                NoMatch => {
181                                    lookup_events.write(LookupEvent::Hide);
182                                    if let Ok(mut ecommands) = commands.get_entity(id) {
183                                        ecommands.try_insert(Feedback::info(format!("{}", e)));
184                                    }
185                                }
186                                Message(s) => {
187                                    lookup_events.write(LookupEvent::Hide);
188                                    if let Ok(mut ecommands) = commands.get_entity(id) {
189                                        ecommands.try_insert(Feedback::info(s));
190                                    }
191                                }
192                                OneMatch(s) => {
193                                    lookup_events.write(LookupEvent::Hide);
194                                    text_state.set_value(&s);
195                                }
196                                ManyMatches => {
197                                    if let Ok(mut ecommands) = commands.get_entity(id) {
198                                        ecommands.try_insert(Feedback::warn("require match"));
199                                    }
200                                    let matches = autocomplete.all_lookups(&text_state.value);
201
202                                    lookup_events.write(LookupEvent::Completions(matches));
203                                    if let Some(new_input) =
204                                        autocomplete.longest_prefix(&text_state.value)
205                                    {
206                                        text_state.set_value(&new_input);
207                                    }
208                                }
209                            }
210                            continue;
211                        }
212                    }
213                    lookup_events.write(LookupEvent::Hide);
214                    commands.trigger(Submit::new(id, Ok(text_state.value.clone())));
215                    block_and_move_focus(&mut commands, id, &nav, &mut input_focus);
216                }
217                Key::Escape => {
218                    commands.trigger(Submit::<String>::new(id, Err(bevy_asky::Error::Cancel)));
219                    if let Ok(mut ecommands) = commands.get_entity(id) {
220                        ecommands.try_insert(Feedback::error("canceled"));
221                    }
222                    block_focus(&mut commands, id);
223                }
224                _x => {
225                    // info!("Unhandled key {x:?}");
226                }
227            }
228        }
229    }
230}

232fn block_and_move_focus(
233    commands: &mut Commands,
234    id: Entity,
235    nav: &TabNavigation,
236    input_focus: &mut InputFocus,
237) {
238    match nav.navigate(input_focus, NavAction::Next) {
239        Ok(next) => input_focus.set(next, FocusCause::Navigated),
240        Err(_) => input_focus.clear(),
241    }
242    commands.entity(id).try_insert(TabIndex(-1));
243}
244
245fn block_focus(commands: &mut Commands, id: Entity) {
246    commands.entity(id).try_insert(TabIndex(-1));

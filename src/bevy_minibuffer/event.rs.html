69/// This holds the last act run.
70///
71/// It is set prior to the command being run, so a command can look up its own
72/// run event and act using this resource.
73#[derive(Resource, Debug, Default, Deref, DerefMut)]
74pub struct LastRunAct(Option<RunActEvent>);
75
76impl LastRunAct {
77    /// Return the hotkey associated with this run.
78    pub fn hotkey(&self, acts: &mut QueryLens<&Act>) -> Option<Hotkey> {
79        self.0.as_ref().and_then(|run_act| run_act.hotkey(acts))
80    }
81}

83impl RunActEvent {
84    /// Make a new run act event.
85    pub fn new(act: ActRef) -> Self {
86        Self {
87            act,
88            hotkey: None,
89            // input: None,
90        }
91    }
92
93    pub fn from_act(act: &Act, id: Entity) -> Self {
94        Self {
95            act: ActRef {
96                id,
97                flags: act.flags,
98            },
99            hotkey: None,
100            // input: None,
101        }
102    }
103
104    /// Set the hotkey index.
105    pub fn with_hotkey(mut self, index: usize) -> Self {
106        self.hotkey = Some(index);
107        self
108    }
109
110    /// Return the hotkey associated with this run.
111    pub fn hotkey(&self, acts: &mut QueryLens<&Act>) -> Option<Hotkey> {
112        acts.query()
113            .get(self.act.id)
114            .ok()
115            .and_then(|act| self.hotkey.map(|index| act.hotkeys[index].clone()))
116    }
117}

149/// Dispatch an event
150///
151/// This event relays another event to fire.
152///
153/// Allows minibuffer to use one channel to dispatch multiple kinds of events.
154#[doc(hidden)]
155#[derive(Debug, Clone, Event, Message)]
156#[allow(private_interfaces)]
157pub enum DispatchEvent {
158    /// Send a look up event.
159    LookupEvent(LookupEvent),
160    /// Send a run act event.
161    RunActEvent(RunActEvent),
162    /// Send a lookup and run act event.
163    RunActByNameEvent(RunActByNameEvent),
164    /// Emit a message.
165    EmitMessage(String),
166    /// Clear the buffer.
167    Clear,
168    /// Show the buffer.
169    SetVisible(bool),
170    /// This event has been "taken" already.
171    Taken,
172}
173
174impl From<LookupEvent> for DispatchEvent {
175    fn from(e: LookupEvent) -> Self {
176        Self::LookupEvent(e)
177    }
178}
179impl From<RunActEvent> for DispatchEvent {
180    fn from(e: RunActEvent) -> Self {
181        Self::RunActEvent(e)
182    }
183}

185pub(crate) fn dispatch_events(
186    mut dispatch_events: MessageReader<DispatchEvent>,
187    mut lookup_events: MessageWriter<LookupEvent>,
188    mut minibuffer: Minibuffer,
189) {
190    use crate::event::DispatchEvent::*;
191    for e in dispatch_events.read() {
192        match e {
193            LookupEvent(l) => {
194                lookup_events.write(l.clone());
195            }
196            RunActEvent(e) => {
197                minibuffer.run_act(e.act);
198            }
199            RunActByNameEvent(e) => {
200                minibuffer.run_act(e.clone().name);
201            }
202            EmitMessage(s) => {
203                minibuffer.message(s.to_string());
204            }
205            Clear => {
206                minibuffer.clear();
207            }
208            SetVisible(show) => {
209                minibuffer.set_visible(*show);
210            }
211            Taken => {}
212        }
213    }
214}

216#[derive(Debug, Reflect)]
217pub enum KeyChordState {
218    Unhandled(KeyChord),
219    Canceled,
220    Handled,
221}
222
223#[derive(EntityEvent, Message, Debug, Reflect)]
224pub struct KeyChordEvent {
225    pub entity: Entity,
226    pub state: KeyChordState,
227}
228
229impl KeyChordEvent {
230    pub fn new(entity: Entity, chord: KeyChord) -> Self {
231        Self {
232            entity,
233            state: KeyChordState::Unhandled(chord),
234        }
235    }
236
237    pub fn canceled(entity: Entity) -> Self {
238        Self {
239            entity,
240            state: KeyChordState::Canceled,
241        }
242    }
243
244    pub fn take(&mut self) -> Result<KeyChord, Error> {
245        match std::mem::replace(&mut self.state, KeyChordState::Handled) {
246            KeyChordState::Unhandled(chord) => Ok(chord),
247            KeyChordState::Handled => Err(Error::Message("Event already handled".into())),
248            KeyChordState::Canceled => Err(bevy_asky::Error::Cancel.into()),
249        }
250    }
251}

253/// Run act for any [RunActEvent].
254pub(crate) fn run_acts(
255    mut events: MessageReader<RunActEvent>,
256    mut commands: Commands,
257    run_act_map: Res<RunActMap>,
258    acts: Query<&Act>,
259    mut last: ResMut<LastRunAct>,
260    mut next_prompt_state: ResMut<NextState<PromptState>>,
261    frame_count: Res<FrameCount>,
262) {
263    for e in events.read() {
264        let act = match acts.get(e.act.id) {
265            Ok(act) => act,
266            Err(err) => {
267                warn!("Could not find act: {err}");
268                continue;
269            }
270        };
271        trace!("act {:?} frame {}", &act, frame_count.0);
272
273        // Set prompt state if needed
274        if e.act.flags.contains(ActFlags::ShowMinibuffer) {
275            next_prompt_state.set(PromptState::Visible);
276        }
277
278        // Run the act
279        let run_act = act
280            .input
281            .as_ref()
282            .and_then(|x| run_act_map.get(x).map(|y| &**y));
283
284        let run_act = run_act.unwrap_or(&ActSystem);
285        last.0 = Some(*e);
286        if let Err(error) = run_act.run(act.system_id, &mut commands) {
287            warn!("Error running act '{}': {:?}", act.name, error);
288        }
289    }
290}

292/// Lookup and run act for any [RunActByNameEvent].
293pub(crate) fn run_acts_by_name(
294    mut events: MessageReader<RunActByNameEvent>,
295    mut commands: Commands,
296    acts: Query<(Entity, &Act)>,
297) {
298    for e in events.read() {
299        if let Some((id, act)) = acts.iter().find(|(_, a)| a.name == e.name) {
300            let new_event = RunActEvent {
301                act: ActRef::from_act(act, id),
302                hotkey: None,
303            };
304            commands.write_message(new_event);
305        } else {
306            warn!("No act named '{}' found.", e.name);
307        }
308    }
309}

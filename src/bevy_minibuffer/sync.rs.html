28/// Manipulate minibuffer synchronously with this [SystemParam].
29///
30/// The "synchronized" version of Minibuffer. It uses Bevy's trigger mechanism
31/// to communicate outcomes. Like many `SystemParams` it cannot be passed into
32/// long-lived closures like async blocks or triggers.
33///
34/// For async blocks, [MinibufferAsync] is available with the "async" feature
35/// flag.
36///
37/// For trigger blocks, one cannot pass [Minibuffer] into the block, but one can
38/// declare a new [Minibuffer] system parameter.
39#[derive(SystemParam)]
40pub struct Minibuffer<'w, 's> {
41    /// The query for where the Minibuffer contents go. Expected to be singular.
42    dest: Query<'w, 's, Entity, With<PromptContainer>>,
43    /// Commands
44    pub commands: Commands<'w, 's>,
45    /// prompt_state
46    prompt_state: Res<'w, State<PromptState>>,
47    /// next prompt state
48    next_prompt_state: ResMut<'w, NextState<PromptState>>,
49    /// macro state
50    pub(crate) tape_recorder: ResMut<'w, TapeRecorder>,
51}

53/// An [EntityCommands] extension trait
54pub trait MinibufferCommands {

56    fn prompt_children<T: Construct + Component + Part>(
57        &mut self,

59    ) -> EntityCommands<'_>
60    where
61        <T as Construct>::Props: Send;
62}

64impl MinibufferCommands for EntityCommands<'_> {
65    fn prompt_children<T: Construct + Component + Part>(
66        &mut self,
67        props: impl IntoIterator<Item = impl Into<T::Props>>,
68    ) -> EntityCommands<'_>
69    where
70        <T as Construct>::Props: Send,
71    {
72        self.construct_children::<Add0<T, View>>(props);

74    }

102    /// Leave a message in the minibuffer.
103    pub fn message(&mut self, msg: impl Into<String>) {
104        let msg = msg.into();
105        let dest = self.dest.single().expect("minibuffer dest");
106        if let Some(mut commands) = Dest::ReplaceChildren(dest).get_entity(&mut self.commands) {
107            commands.insert(Text::new(msg)).insert(TextLayout {
108                linebreak: LineBreak::WordOrCharacter,
109                ..default()
110            });
111        }
112    }

158    /// Read input from user with autocomplete provided by a [Lookup].
159    pub fn prompt_lookup<L>(
160        &mut self,
161        prompt: impl Into<<TextField as Construct>::Props>,
162        lookup: L,
163    ) -> EntityCommands<'_>
164    where
165        L: Lookup + Send + Sync + 'static,
166    {
167        let dest = self.dest.single().expect("minibuffer dest");
168        let commands = Dest::ReplaceChildren(dest).entity(&mut self.commands);
169        let autocomplete = AutoComplete::new(lookup);
170        autocomplete.construct(commands, prompt)
171    }
172
173    /// Read input from user that maps to other another type.
174    ///
175    /// Instead of triggering [`Submit<String>`] it will trigger [`Completed<T>`].
176    pub fn prompt_map<L>(
177        &mut self,
178        prompt: impl Into<<TextField as Construct>::Props>,
179        lookup: L,
180    ) -> EntityCommands<'_>
181    where
182        L: Lookup + Clone + LookupMap + Send + Sync + 'static,
183        <L as LookupMap>::Item: Sync + Debug,
184    {
185        let dest = self.dest.single().expect("minibuffer dest");
186        let commands = Dest::ReplaceChildren(dest).entity(&mut self.commands);
187        let autocomplete = AutoComplete::new(lookup.clone());
188        let mut ecommands = autocomplete.construct(commands, prompt);
189        ecommands
190            .insert(RequireMatch)
191            // TODO: We should probably return the input string in either case.
192            .observe(
193                move |mut trigger: On<Submit<String>>, mut commands: Commands| {
194                    let entity = trigger.event().entity;
195                    let mut input = None;
196                    let result: Result<L::Item, Error> = trigger
197                        .event_mut()
198                        .take_result()
199                        .map_err(Error::from)
200                        .and_then(|s| {
201                            let r = lookup.resolve_res(&s).map_err(Error::from);
202                            input = Some(s);
203                            r
204                        });
205                    commands.trigger(Completed::new(entity, result, input));
206                },
207            );
208        ecommands
209    }

211    /// Clear the minibuffer.
212    pub fn clear(&mut self) {
213        let dest = self.dest.single().expect("minibuffer dest");
214        self.commands.entity(dest).despawn_related::<Children>();
215    }

217    /// Return the visible state.

222    /// Set the visible state.
223    pub fn set_visible(&mut self, show: bool) {
224        self.next_prompt_state.set(if show {
225            PromptState::Visible
226        } else {
227            PromptState::Invisible
228        });

61/// Create the UI layout.
62fn spawn_layout(mut commands: Commands) {
63    commands
64        .spawn((
65            Node {
66                position_type: PositionType::Absolute,
67                bottom: Val::Px(0.0),
68                right: Val::Px(0.0),
69                left: Val::Px(0.0),
70                flex_direction: FlexDirection::Column,
71                ..default()
72            },
73            Name::new("minibuffer"),
74            MinibufferNode,
75        ))
76        .with_children(|builder| {
77            builder.spawn((Name::new("acts"), ActContainer));
78            builder
79                .spawn((
80                    Node {
81                        flex_direction: FlexDirection::Row,
82                        ..default()
83                    },
84                    Visibility::Hidden,
85                ))
86                .insert(Name::new("completions"))
87                .with_children(|builder| {
88                    // List with hidden overflow
89                    builder
90                        .spawn((
91                            Node {
92                                flex_direction: FlexDirection::Column,
93                                align_self: AlignSelf::FlexEnd,
94                                // height: Val::Percent(50.),
95                                min_width: Val::Percent(25.),
96                                overflow: Overflow::clip_y(),
97                                ..default()
98                            },
99                            BackgroundColor(Color::srgb(0.10, 0.10, 0.10)),
100                        ))
101                        .insert(CompletionContainer)
102                        .with_children(|builder| {
103                            builder
104                                .spawn((
105                                    Node {
106                                            flex_direction: FlexDirection::Column,
107                                            align_items: AlignItems::FlexStart,
108                                            flex_grow: 0.,
109                                            padding: UiRect {
110                                                top: PADDING,
111                                                left: LEFT_PADDING,
112                                                right: PADDING * 2.,
113                                                bottom: PADDING,
114                                            },
115                                            margin: UiRect {
116                                                bottom: PADDING,
117                                                ..default()
118                                            },
119                                            ..default()
120                                    },
121                                    BackgroundColor(Color::BLACK),
122                                    ScrollingList::default(),
123                                    AccessibilityNode(Accessible::new(Role::List)),
124                                ))
125                                // .with_children(|parent| {
126                                //     // List items
127                                //     for i in 0..30 {
128                                //         parent.spawn(completion_item(
129                                //             format!("Item {i}"),
130                                //             TextStyle::default(),
131                                //         ));
132                                //     }
133                                // })
134                                ;
135
136                            builder.spawn(Node::default());
137                        });
138                });
139            builder
140                .spawn((
141                    Node::default(),
142                    BackgroundColor(Color::BLACK),
143                    Visibility::Hidden,
144                    Name::new("bar"),
145                    BottomBar,
146                ))
147                .with_children(|parent| {
148                    parent.spawn((
149                        Node {
150                            flex_wrap: FlexWrap::Wrap,
151                            flex_direction: FlexDirection::Row,
152                            flex_grow: 1.,
153                            padding: UiRect {
154                                top: PADDING,
155                                left: LEFT_PADDING,
156                                right: PADDING,
157                                bottom: PADDING,
158                            },
159                            ..default()
160                        },
161                        Name::new("buffer"),
162                        PromptContainer,
163                    ));
164                    parent.spawn((
165                        Node {
166                            // TODO: How do I get the height of our default
167                            // font? It appears to be 30, but I'd like to do
168                            // better than hard coded.
169                            height: Val::Px(30.0),
170                            ..default()
171                        },
172                        Name::new("icons"),
173                        IconContainer,
174                    ));
175                });
176        });
177}
